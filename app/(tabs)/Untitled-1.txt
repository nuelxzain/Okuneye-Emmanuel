import React, { useState, useCallback } from 'react';
import {
  StyleSheet,
  Text,
  View,
  TouchableOpacity,
  StatusBar,
  Dimensions,
  Platform,
  SafeAreaView,
} from 'react-native';

const { width } = Dimensions.get('window');
const PADDING = 20;
const GAP = 12;
const BTN_SIZE = (width - PADDING * 2 - GAP * 3) / 4;

// --- Types ---
type BtnType = 'num' | 'op' | 'action' | 'eq' | 'zero';
type Btn = { label: string; type: BtnType };

// --- Button Layout ---
const ROWS: Btn[][] = [
  [
    { label: 'AC', type: 'action' },
    { label: '+/-', type: 'action' },
    { label: '%', type: 'action' },
    { label: '÷', type: 'op' },
  ],
  [
    { label: '7', type: 'num' },
    { label: '8', type: 'num' },
    { label: '9', type: 'num' },
    { label: '×', type: 'op' },
  ],
  [
    { label: '4', type: 'num' },
    { label: '5', type: 'num' },
    { label: '6', type: 'num' },
    { label: '−', type: 'op' },
  ],
  [
    { label: '1', type: 'num' },
    { label: '2', type: 'num' },
    { label: '3', type: 'num' },
    { label: '+', type: 'op' },
  ],
  [
    { label: '0', type: 'zero' },
    { label: '.', type: 'num' },
    { label: '=', type: 'eq' },
  ],
];

// --- Helpers ---
function calculate(a: number, op: string, b: number): number {
  switch (op) {
    case '+': return a + b;
    case '−': return a - b;
    case '×': return a * b;
    case '÷': return b !== 0 ? a / b : 0;
    default: return b;
  }
}

function fmt(val: string): string {
  const n = parseFloat(val);
  if (isNaN(n)) return '0';
  // Use toPrecision or toFixed to handle floating point jitters like 0.1 + 0.2
  return parseFloat(n.toPrecision(12)).toString();
}

export default function App() {
  const [display, setDisplay] = useState('0');
  const [prev, setPrev] = useState<string | null>(null);
  const [op, setOp] = useState<string | null>(null);
  const [activeOp, setActiveOp] = useState<string | null>(null);
  const [expr, setExpr] = useState('');
  const [waiting, setWaiting] = useState(false);

  // --- Logic Handlers ---
  const inputDigit = useCallback((d: string) => {
    setDisplay(cur => {
      if (waiting) {
        setWaiting(false);
        return d === '.' ? '0.' : d;
      }
      if (cur === '0' && d !== '.') return d;
      if (d === '.' && cur.includes('.')) return cur;
      return cur + d;
    });
  }, [waiting]);

  const inputOperator = useCallback((nextOp: string) => {
    const value = parseFloat(display);

    if (activeOp && waiting) {
      setOp(nextOp);
      setActiveOp(nextOp);
      setExpr(prev + ' ' + nextOp);
      return;
    }

    if (prev === null) {
      setPrev(display);
    } else if (op) {
      const result = calculate(parseFloat(prev), op, value);
      const fmtResult = fmt(String(result));
      setDisplay(fmtResult);
      setPrev(fmtResult);
    }

    setWaiting(true);
    setOp(nextOp);
    setActiveOp(nextOp);
    setExpr(display + ' ' + nextOp);
  }, [display, prev, op, waiting]);

  const inputEquals = useCallback(() => {
    if (!op || prev === null) return;
    const result = calculate(parseFloat(prev), op, parseFloat(display));
    const fmtResult = fmt(String(result));
    
    setExpr('');
    setDisplay(fmtResult);
    setPrev(null);
    setOp(null);
    setActiveOp(null);
    setWaiting(true);
  }, [display, prev, op]);

  const inputAction = useCallback((action: string) => {
    if (action === 'AC') {
      setDisplay('0'); setPrev(null); setOp(null);
      setActiveOp(null); setExpr(''); setWaiting(false);
    } else if (action === '+/-') {
      setDisplay(d => (parseFloat(d) * -1).toString());
    } else if (action === '%') {
      setDisplay(d => (parseFloat(d) / 100).toString());
    }
  }, []);

  const onPress = useCallback((btn: Btn) => {
    switch (btn.type) {
      case 'num':
      case 'zero':   inputDigit(btn.label); break;
      case 'op':     inputOperator(btn.label); break;
      case 'eq':     inputEquals(); break;
      case 'action': inputAction(btn.label); break;
    }
  }, [inputDigit, inputOperator, inputEquals, inputAction]);

  // --- Dynamic Styles ---
  const getBtnBg = (btn: Btn) => {
    if (btn.type === 'action') return '#A5A5A5';
    if (btn.type === 'op' || btn.type === 'eq') {
      return btn.label === activeOp ? '#FFFFFF' : '#FF9F0A';
    }
    return '#333333';
  };

  const getBtnTextColor = (btn: Btn) => {
    if (btn.type === 'action') return '#000000';
    if (btn.type === 'op' && btn.label === activeOp) return '#FF9F0A';
    return '#FFFFFF';
  };

  return (
    <SafeAreaView style={styles.container}>
      <StatusBar barStyle="light-content" />
      
      <View style={styles.displayContainer}>
        <Text style={styles.expressionText}>{expr}</Text>
        <Text numberOfLines={1} adjustsFontSizeToFit style={styles.displayText}>
          {display}
        </Text>
      </View>

      <View style={styles.keypad}>
        {ROWS.map((row, i) => (
          <View key={i} style={styles.row}>
            {row.map((btn) => (
              <TouchableOpacity
                key={btn.label}
                onPress={() => onPress(btn)}
                activeOpacity={0.8}
                style={[
                  styles.btn,
                  {
                    backgroundColor: getBtnBg(btn),
                    width: btn.type === 'zero' ? BTN_SIZE * 2 + GAP : BTN_SIZE,
                    alignItems: btn.type === 'zero' ? 'flex-start' : 'center',
                    paddingLeft: btn.type === 'zero' ? 30 : 0,
                  }
                ]}
              >
                <Text style={[styles.btnText, { color: getBtnTextColor(btn) }]}>
                  {btn.label}
                </Text>
              </TouchableOpacity>
            ))}
          </View>
        ))}
      </View>
    </SafeAreaView>
  );
}

const styles = StyleSheet.create({
  container: {
    flex: 1,
    backgroundColor: '#000',
    justifyContent: 'flex-end',
  },
  displayContainer: {
    paddingHorizontal: 30,
    paddingBottom: 20,
    alignItems: 'flex-end',
  },
  expressionText: {
    color: '#777',
    fontSize: 20,
    height: 30,
  },
  displayText: {
    color: '#fff',
    fontSize: 80,
    fontWeight: '300',
  },
  keypad: {
    paddingHorizontal: PADDING,
    paddingBottom: 40,
    gap: GAP,
  },
  row: {
    flexDirection: 'row',
    justifyContent: 'space-between',
  },
  btn: {
    height: BTN_SIZE,
    borderRadius: BTN_SIZE / 2,
    justifyContent: 'center',
  },
  btnText: {
    fontSize: 32,
    fontWeight: '400',
  },
});
